Quick Start 

# Install
pip install flaxon

# Create app.py
echo 'from flaxon import Flaxon
app = Flaxon("hello")
@app.get("/")
async def home():
    return {"message": "Hello!"}' > app.py

# Run
flaxon run app:app --reload

# Open http://localhost:8000
